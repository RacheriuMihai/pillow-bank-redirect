  <meta charset="utf-8">

  <meta name="referrer" content="no-referrer">
  <title>Pillow · returning from your bank…</title>
  <style>
    body { font-family: -apple-system, "Segoe UI", Roboto, sans-serif; color: #1d2733; background: #f6f8fa;
           display: grid; place-items: center; min-height: 100vh; margin: 0; }

<body>
  <div class="box">
    <h1>Pillow</h1>
    <p>Returning you to the app…</p>
    <p id="fallback" hidden>If nothing happens, <a id="link" href="#">continue here</a>.
      The Pillow API must be running on this computer.</p>
  </div>
  <script>
    var API_CALLBACK = "http://localhost:8000/bank/callback";
    var incoming = new URLSearchParams(window.location.search);
    var forward = new URLSearchParams();
    ["code", "state", "error"].forEach(function (key) {
      if (incoming.has(key)) forward.set(key, incoming.get(key));
    });
    var target = API_CALLBACK + "?" + forward.toString();
    document.getElementById("link").href = target;
    document.getElementById("fallback").hidden = false;
    window.location.replace(target);
  </script>
</body>
